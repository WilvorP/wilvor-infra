# ============================================================
# 1. MOVE INTO TERRAFORM DEV ENVIRONMENT
# ============================================================

cd envs/dev


# ============================================================
# 2. INITIALIZE TERRAFORM
# ============================================================

terraform init


# ============================================================
# 3. FORMAT AND VALIDATE TERRAFORM
# ============================================================

terraform fmt -recursive

terraform validate


# ============================================================
# 4. REVIEW PLAN
# ============================================================

terraform plan

# Optional but better: save the plan
terraform plan -out=tfplan


# ============================================================
# 5. APPLY INFRASTRUCTURE
# ============================================================

terraform apply tfplan


# ============================================================
# 6. CONFIRM OUTPUTS
# ============================================================

terraform output

terraform output -raw aircraft_raw_stream_name
terraform output -raw aircraft_clean_stream_name
terraform output -raw aircraft_archive_bucket_name
terraform output -raw opensky_credentials_secret_arn
terraform output -raw opensky_credentials_secret_name
terraform output -raw aircraft_raw_processor_lambda_name


# ============================================================
# 7. UPLOAD OPENSKY CREDENTIALS TO AWS SECRETS MANAGER
# ============================================================
# This assumes your local file exists at:
# .secrets/credentials.json
#
# Expected JSON format:
# {
#   "clientId": "your-client-id",
#   "clientSecret": "your-client-secret"
# }

$secretArn = terraform output -raw opensky_credentials_secret_arn
$secretFile = (Resolve-Path ..\..\.secrets\credentials.json).Path

aws secretsmanager put-secret-value `
  --profile wilvor-dev `
  --region us-west-1 `
  --secret-id $secretArn `
  --secret-string file://$secretFile


# Optional: verify the secret can be read
aws secretsmanager get-secret-value `
  --profile wilvor-dev `
  --region us-west-1 `
  --secret-id $secretArn


# ============================================================
# 8. VERIFY RAW PROCESSOR LAMBDA + EVENT SOURCE MAPPING
# ============================================================

aws lambda get-function `
  --function-name wilvor-dev-aircraft-raw-processor `
  --profile wilvor-dev `
  --region us-west-1

aws lambda list-event-source-mappings `
  --function-name wilvor-dev-aircraft-raw-processor `
  --profile wilvor-dev `
  --region us-west-1

# Expected:
# State: Enabled
# StartingPosition: LATEST
# LastProcessingResult may say "No records processed" until you run the local poller.


# ============================================================
# 9. RETURN TO REPO ROOT
# ============================================================

cd ../..

# ============================================================
# 14. VERIFY KINESIS RAW STREAM EXISTS
# ============================================================

cd envs/dev

$rawStreamName = terraform output -raw aircraft_raw_stream_name

aws kinesis describe-stream-summary `
  --profile wilvor-dev `
  --region us-west-1 `
  --stream-name $rawStreamName

cd ../..


# ============================================================
# 15. VERIFY KINESIS CLEAN STREAM EXISTS
# ============================================================

cd envs/dev

$cleanStreamName = terraform output -raw aircraft_clean_stream_name

aws kinesis describe-stream-summary `
  --profile wilvor-dev `
  --region us-west-1 `
  --stream-name $cleanStreamName

cd ../..


# ============================================================
# 16. OPTIONAL: READ A FEW RECORDS FROM CLEAN STREAM
# ============================================================
# This confirms the processor published normalized aircraft_current_state.v1 records.

cd envs/dev

$cleanStreamName = terraform output -raw aircraft_clean_stream_name

$shardId = aws kinesis describe-stream `
  --profile wilvor-dev `
  --region us-west-1 `
  --stream-name $cleanStreamName `
  --query "StreamDescription.Shards[0].ShardId" `
  --output text

$iterator = aws kinesis get-shard-iterator `
  --profile wilvor-dev `
  --region us-west-1 `
  --stream-name $cleanStreamName `
  --shard-id $shardId `
  --shard-iterator-type TRIM_HORIZON `
  --query "ShardIterator" `
  --output text

aws kinesis get-records `
  --profile wilvor-dev `
  --region us-west-1 `
  --shard-iterator $iterator `
  --limit 5

cd ../..


# ============================================================
