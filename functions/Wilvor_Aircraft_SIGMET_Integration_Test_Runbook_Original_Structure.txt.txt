$start30m = (Get-Date).AddMinutes(-30).ToUniversalTime().ToString("yyyy-MM-ddTHH:mm:ssZ")
$endNow = (Get-Date).ToUniversalTime().ToString("yyyy-MM-ddTHH:mm:ssZ")

foreach ($productType in @("METAR", "TAF", "SIGMET")) {
  Write-Host "===== WEATHER EVENTS: $productType ====="

  aws cloudwatch get-metric-statistics `
    --profile wilvor-dev `
    --region us-west-1 `
    --namespace "Wilvor/WeatherEvents" `
    --metric-name "WeatherChangedEvents" `
    --dimensions Name=ProductType,Value=$productType `
    --start-time $start30m `
    --end-time $endNow `
    --period 60 `
    --statistics Sum `
    --output table
}
